-- D7  Core/PaperA/Defs/LatticeShift.lean
/-- The link shift `T_j(U)` of Paper A, (2.3): `(T_j(U) f)(x) = conj(U_{x j}) f(x + e_j)` on the
torus `X = (ℤ/nℤ)³`, where `e_j = Pi.single j 1` and `U_{x j}` is the link variable on the link
from `x` to `x + e_j`. -/
def latticeShift (n : ℕ) (U : (Fin 3 → ZMod n) → Fin 3 → ℂ) (j : Fin 3) :
    Matrix (Fin 3 → ZMod n) (Fin 3 → ZMod n) ℂ :=
  Matrix.of fun x y => if y = x + Pi.single j 1 then star (U x j) else 0

-- D8  Core/PaperA/Defs/WilsonHopping.lean
/-- `v_j(U) = (T_j(U) - T_j(U)^*) / (2 i a)` (Paper A, (2.3)). -/
noncomputable def wilsonHopping (n : ℕ) (a : ℝ) (U : (Fin 3 → ZMod n) → Fin 3 → ℂ) (j : Fin 3) :
    Matrix (Fin 3 → ZMod n) (Fin 3 → ZMod n) ℂ :=
  (2 * Complex.I * (a : ℂ))⁻¹ • (latticeShift n U j - (latticeShift n U j)ᴴ)

-- D9  Core/PaperA/Defs/WilsonMass.lean
/-- The Wilson mass `M(U) = m + (r / 2a) Σ_j (2 - T_j(U) - T_j(U)^*)` (Paper A, (2.3)). -/
noncomputable def wilsonMass (n : ℕ) (a m r : ℝ) (U : (Fin 3 → ZMod n) → Fin 3 → ℂ) :
    Matrix (Fin 3 → ZMod n) (Fin 3 → ZMod n) ℂ :=
  (m : ℂ) • 1 +
    ((r / (2 * a) : ℝ) : ℂ) • ∑ j, ((2 : ℂ) • 1 - latticeShift n U j - (latticeShift n U j)ᴴ)

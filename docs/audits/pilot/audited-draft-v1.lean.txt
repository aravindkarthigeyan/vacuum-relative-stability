-- D1  Core/PaperA/Defs/Step.lean
/-- One step of an ordered history (Paper A, Lemma 3.1): a free evolution of duration `τ`
generated by `Σ_j α_j v_j + β M` with spatial coefficients `v` and mass matrix `M`, or a
spin-scalar kick `exp (i Θ)` with spatial generator `Θ`. -/
inductive Step (ι : Type*) where
  | free (τ : ℝ) (v : Fin 3 → Matrix ι ι ℂ) (M : Matrix ι ι ℂ)
  | kick (Θ : Matrix ι ι ℂ)

-- D2  Core/PaperA/Defs/DiracOperator.lean
/-- `h = Σ_j α_j v_j + β M` on `ℓ²(ι) ⊗ ℂ⁴`, the spatial factor first
(Paper A, (2.3) and Lemma 3.1). -/
def diracOperator {ι : Type*} (α : Fin 3 → Matrix (Fin 4) (Fin 4) ℂ)
    (β : Matrix (Fin 4) (Fin 4) ℂ) (v : Fin 3 → Matrix ι ι ℂ) (M : Matrix ι ι ℂ) :
    Matrix (ι × Fin 4) (ι × Fin 4) ℂ :=
  ∑ j, v j ⊗ₖ α j + M ⊗ₖ β

-- D3  Core/PaperA/Defs/StepGenerator.lean
/-- The integral `∫ A dt` of the generator `A = c h + i Φ` of `U' = -A U` over one step:
`τ h` for a free step, and `-i (Θ ⊗ 1)` for the kick `exp (i (Θ ⊗ 1))`. -/
def Step.generator {ι : Type*} (α : Fin 3 → Matrix (Fin 4) (Fin 4) ℂ)
    (β : Matrix (Fin 4) (Fin 4) ℂ) : Step ι → Matrix (ι × Fin 4) (ι × Fin 4) ℂ
  | .free τ v M => (τ : ℂ) • diracOperator α β v M
  | .kick Θ => -(Complex.I • (Θ ⊗ₖ (1 : Matrix (Fin 4) (Fin 4) ℂ)))

-- D4  Core/PaperA/Defs/StepFreeTime.lean
/-- Free time of a step: `τ` for a free step, `0` for a kick. -/
def Step.freeTime {ι : Type*} : Step ι → ℝ
  | .free τ _ _ => τ
  | .kick _ => 0

-- D5  Core/PaperA/Defs/TransferMatrix.lean
/-- The one-particle transfer matrix `U` of a history: the ordered product of the step
propagators `exp (-∫A)`, later steps on the left. -/
noncomputable def transferMatrix {ι : Type*} [Fintype ι] [DecidableEq ι]
    (α : Fin 3 → Matrix (Fin 4) (Fin 4) ℂ) (β : Matrix (Fin 4) (Fin 4) ℂ)
    (hist : List (Step ι)) : Matrix (ι × Fin 4) (ι × Fin 4) ℂ :=
  (hist.map fun s => NormedSpace.exp (-(s.generator α β))).reverse.prod

-- D6  Core/PaperA/Defs/CenteredTrace.lean
/-- The centered finite-CAR trace `F = exp (½ ∫ Tr A) det (I + U)` (Paper A, (3.1)). -/
noncomputable def centeredTrace {ι : Type*} [Fintype ι] [DecidableEq ι]
    (α : Fin 3 → Matrix (Fin 4) (Fin 4) ℂ) (β : Matrix (Fin 4) (Fin 4) ℂ)
    (hist : List (Step ι)) : ℂ :=
  Complex.exp ((1 / 2 : ℂ) * (hist.map fun s => (s.generator α β).trace).sum) *
    (1 + transferMatrix α β hist).det

-- T1  Core/PaperA/Frozen/PositivityAndTransferAnnulus.lean   (lem:transfer)
theorem positivity_and_transfer_annulus
    {ι : Type*} [Fintype ι] [DecidableEq ι]
    (α : Fin 3 → Matrix (Fin 4) (Fin 4) ℂ) (β : Matrix (Fin 4) (Fin 4) ℂ)
    (hα_herm : ∀ j, (α j).IsHermitian) (hβ_herm : β.IsHermitian)
    (hα_sq : ∀ j, α j * α j = 1) (hβ_sq : β * β = 1)
    (hα_anti : ∀ j k, j ≠ k → α j * α k + α k * α j = 0)
    (hαβ_anti : ∀ j, α j * β + β * α j = 0)
    (d : ℕ) (hd : 4 * Fintype.card ι = 2 * d)
    (mStar b : ℝ) (hm : 0 < mStar) (hb : 0 < b)
    (hist : List (Step ι))
    (hfree : ∀ τ v M, Step.free τ v M ∈ hist →
      0 ≤ τ ∧ (∀ j, (v j).IsHermitian) ∧ (M - (mStar : ℂ) • 1).PosSemidef)
    (hkick : ∀ Θ, Step.kick Θ ∈ hist → Θ.IsHermitian)
    (htime : (hist.map Step.freeTime).sum = b) :
    (0 < (centeredTrace α β hist).re ∧ (centeredTrace α β hist).im = 0) ∧
    ((transferMatrix α β hist).charpoly.roots.filter
        (fun z => ‖z‖ ≤ Real.exp (-(b * mStar)))).card = d ∧
    (transferMatrix α β hist).charpoly.roots.filter
        (fun z => ¬ ‖z‖ ≤ Real.exp (-(b * mStar))) =
      ((transferMatrix α β hist).charpoly.roots.filter
        (fun z => ‖z‖ ≤ Real.exp (-(b * mStar)))).map (fun z => (star z)⁻¹) ∧
    ∀ z ∈ (transferMatrix α β hist).charpoly.roots,
      ¬ ‖z‖ ≤ Real.exp (-(b * mStar)) → Real.exp (b * mStar) ≤ ‖z‖ := by
  sorry

-- T2  Core/PaperA/Frozen/CanonicalFourierBound.lean   (lem:fourier)
theorem canonical_fourier_bound
    {ι : Type*} [Fintype ι] [DecidableEq ι]
    (α : Fin 3 → Matrix (Fin 4) (Fin 4) ℂ) (β : Matrix (Fin 4) (Fin 4) ℂ)
    (hα_herm : ∀ j, (α j).IsHermitian) (hβ_herm : β.IsHermitian)
    (hα_sq : ∀ j, α j * α j = 1) (hβ_sq : β * β = 1)
    (hα_anti : ∀ j k, j ≠ k → α j * α k + α k * α j = 0)
    (hαβ_anti : ∀ j, α j * β + β * α j = 0)
    (d : ℕ) (hd : 4 * Fintype.card ι = 2 * d)
    (mStar b : ℝ) (hm : 0 < mStar) (hb : 0 < b)
    (hist : List (Step ι))
    (hfree : ∀ τ v M, Step.free τ v M ∈ hist →
      0 ≤ τ ∧ (∀ j, (v j).IsHermitian) ∧ (M - (mStar : ℂ) • 1).PosSemidef)
    (hkick : ∀ Θ, Step.kick Θ ∈ hist → Θ.IsHermitian)
    (htime : (hist.map Step.freeTime).sum = b)
    (q : ℤ) (hq : |q| ≤ d) :
    let F : ℝ → ℂ := fun φ => centeredTrace α β (hist ++ [Step.kick ((φ : ℂ) • 1)])
    let Fhat : ℤ → ℂ := fun p =>
      (2 * Real.pi : ℂ)⁻¹ *
        ∫ φ in (0 : ℝ)..2 * Real.pi, Complex.exp (-(p * φ * Complex.I)) * F φ
    (Fhat 0).im = 0 ∧
      ‖Fhat q‖ ≤ ((2 * d).choose (d + q.natAbs) : ℝ) *
        Real.exp (-(b * mStar * |(q : ℝ)|)) * (Fhat 0).re := by
  sorry
